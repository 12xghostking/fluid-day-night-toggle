React / Next.js Component 
import React from 'react';
import './DayNightToggle.css';

export function DayNightToggle({ isDark, onToggle, size = 'default' }) {
  return (
    <label className={`day-night-toggle size-${size}`}>
      <input
        type="checkbox"
        className="toggle-input"
        role="switch"
        aria-checked={isDark}
        checked={isDark}
        onChange={(e) => onToggle?.(e.target.checked)}
      />
      <div className="toggle-track">
        <div className="sky sky-day">{/* Layered clouds */}</div>
        <div className="sky sky-night">{/* Sparkles & Stars */}</div>
        <div className="cavity-overlay" />
        <div className="toggle-thumb">
          <div className="sun-body" />
          <div className="moon-body">
            <div className="crater crater-1" />
            <div className="crater crater-2" />
            <div className="crater crater-3" />
          </div>
        </div>
      </div>
    </label>
  );
}
Vue 3 Single File Component 
<template>
  <label class="day-night-toggle" :class="`size-${size}`">
    <input
      type="checkbox"
      class="toggle-input"
      role="switch"
      :aria-checked="modelValue"
      :checked="modelValue"
      @change="$emit('update:modelValue', $event.target.checked)"
    />
    <div class="toggle-track">
      <div class="sky sky-day">{/* clouds */}</div>
      <div class="sky sky-night">{/* stars */}</div>
      <div class="cavity-overlay" />
      <div class="toggle-thumb">
        <div class="sun-body" />
        <div class="moon-body">
          <div class="crater crater-1" />
          <div class="crater crater-2" />
          <div class="crater crater-3" />
        </div>
      </div>
    </div>
  </label>
</template>

<script setup>
defineProps({
  modelValue: { type: Boolean, default: false },
  size: { type: String, default: 'default' }
});
defineEmits(['update:modelValue']);
</script>
